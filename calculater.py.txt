    return a * b

    if b == 0:
        print("float division by zero")
        return None
    return a / b
def power(a, b):
    return a ** b
def remainder(a, b):
    return a % b

def select_op(choice):
    if choice == "#":
        return -1
    elif choice == "$":
        return 0
    elif choice in ("+", "-", "/", "*", "^", "%"):

        print(num1_str)

            return 0

        a = float(num1_str)
        num2_str = input("Enter second number: ")
        print(num2_str)
        if num2_str.endswith("$"):
            return 0
        if num2_str.endswith("#"):
            return -1
        b = float(num2_str)
        if choice == "+":
            result = add(a, b)
            print(f"{a} + {b} = {result}")
        elif choice == "-":
            result = subtract(a, b)
            print(f"{a} - {b} = {result}")
        elif choice == "*":
            result = multiply(a, b)
            print(f"{a} * {b} = {result}")
        elif choice == "/":
            result = divide(a, b)
            print(f"{a} / {b} = {result}")
        elif choice == "^":
            result = power(a, b)
            print(f"{a} ^ {b} = {result}")
        elif choice == "%":
            result = remainder(a, b)
            print(f"{a} % {b} = {result}")
    else:
        print("unrecognized operation")
